function positionSurveySignedIn() {
  return request.auth != null
    && request.auth.token.firebase.sign_in_provider == 'google.com';
}
function positionSurveyCoach() {
  return positionSurveySignedIn()
    && request.auth.uid == 'REPLACE_WITH_VERIFIED_COACH_UID';
}
function validPositionPreference() {
  let d = request.resource.data;
  return d.keys().hasOnly(['name', 'first', 'second', 'stable', 'flexible', 'note', 'updatedAt'])
    && d.keys().hasAll(['name', 'first', 'second', 'stable', 'flexible', 'note', 'updatedAt'])
    && d.name is string && d.name.size() > 0 && d.name.size() <= 100
    && d.name.matches('^[^/]+$')
    && exists(/databases/$(database)/documents/players/$(d.name))
    && d.first in ['GK','LB','CB','RB','DM','CM','AM','LW','RW','FW']
    && d.second in ['GK','LB','CB','RB','DM','CM','AM','LW','RW','FW']
    && d.stable is bool && d.flexible is bool
    && d.note is string && d.note.size() <= 500
    && d.updatedAt == request.time;
}
match /privatePositionPreferences/{uid} {
  allow get: if positionSurveyCoach()
    || (positionSurveySignedIn() && request.auth.uid == uid);
  allow list: if positionSurveyCoach();
  allow create: if positionSurveySignedIn() && request.auth.uid == uid
    && validPositionPreference();
  allow update: if positionSurveySignedIn() && request.auth.uid == uid
    && request.resource.data.name == resource.data.name
    && validPositionPreference();
  allow delete: if false;
}
